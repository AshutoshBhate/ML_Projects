import os

os.environ["KERAS_BACKEND"] = "tensorflow"

import keras

# In[2]:
IMAGE_SIZE = 256
BATCH_SIZE = 32
SEED = 12


# In[3]:
dataset = keras.utils.image_dataset_from_directory(
    "PlantVillage",
    shuffle=False,
    image_size=(IMAGE_SIZE, IMAGE_SIZE),
    batch_size=None,
)

class_names = dataset.class_names

dataset = dataset.shuffle(len(dataset), seed=SEED, reshuffle_each_iteration=False)

# In[16]:
total_images = len(dataset)
train_size = int(0.8 * total_images)
validation_size = int(0.1 * total_images)

train_dataset = dataset.take(train_size)
validation_dataset = dataset.skip(train_size).take(validation_size)
test_dataset = dataset.skip(train_size + validation_size)

print("Total images:", total_images)
print("Training images:", len(train_dataset))
print("Validation images:", len(validation_dataset))
print("Test images:", len(test_dataset))


# In[17]:
train_dataset = (
    train_dataset
    .cache()
    .shuffle(1000)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)
validation_dataset = validation_dataset.cache().batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
test_dataset = test_dataset.cache().batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

print("Training batches per epoch:", len(train_dataset))
print("Validation batches:", len(validation_dataset))
print("Test batches:", len(test_dataset))


# In[18]:
resize_rescale = keras.Sequential(
    [
        layers.Resizing(IMAGE_SIZE, IMAGE_SIZE),
        layers.Rescaling(1.0 / 255),
    ],
    name="resize_rescale",
)

data_augmentation = keras.Sequential(
    [
        layers.RandomFlip("horizontal_and_vertical"),
        layers.RandomRotation(0.2),
    ],
    name="data_augmentation",
)


# In[19]:
n_classes = len(class_names)

model = keras.Sequential(
    [
        keras.Input(shape=(IMAGE_SIZE, IMAGE_SIZE, 3)),
        resize_rescale,
        data_augmentation,
        layers.Conv2D(32, (3, 3), activation="relu"),
        layers.MaxPooling2D((2, 2)),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D((2, 2)),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D((2, 2)),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D((2, 2)),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D((2, 2)),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D((2, 2)),
        layers.Flatten(),
        layers.Dense(64, activation="relu"),
        layers.Dense(n_classes, activation="softmax"),
    ],
    name="potato_disease_cnn",
)

model.compile(
    optimizer="adam",
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=False),
    metrics=["accuracy"],
)

model.summary()


# In[20]:
early_stopping = keras.callbacks.EarlyStopping(
    monitor="val_accuracy", patience=8, restore_best_weights=True
)

history = model.fit(
    train_dataset,
    epochs=40,
    verbose=1,
    validation_data=validation_dataset,
    callbacks=[early_stopping],
)


# In[21]:
test_loss, test_accuracy = model.evaluate(test_dataset, verbose=0)
print("Test loss:", round(test_loss, 4))
print("Test accuracy:", round(test_accuracy, 4))


# In[22]:
val_accuracy_per_epoch = history.history["val_accuracy"]
best_epoch = val_accuracy_per_epoch.index(max(val_accuracy_per_epoch)) + 1

print("Training settings:", history.params)
print("Metrics recorded:", list(history.history.keys()))
print("Epochs trained:", len(val_accuracy_per_epoch))
print("Best epoch, whose weights were restored:", best_epoch)
print("Best validation accuracy:", round(max(val_accuracy_per_epoch), 4))

# In[24]:
for image_batch, label_batch in test_dataset.take(1):
    first_image = image_batch[0].numpy().astype("uint8")
    first_label = label_batch[0]
    batch_prediction = model.predict(image_batch, verbose=0)

    plt.imshow(first_image)
    plt.axis("off")
    print("Actual label:", class_names[first_label])
    print("Predicted probabilities:", batch_prediction[0])
    print("Predicted label:", class_names[batch_prediction[0].argmax()])


# In[25]:
def predict(image):
    image_batch = keras.ops.expand_dims(image, 0)
    predictions = model.predict(image_batch, verbose=0)
    predicted_class = class_names[predictions[0].argmax()]
    confidence = round(100 * float(predictions[0].max()), 2)
    return predicted_class, confidence
